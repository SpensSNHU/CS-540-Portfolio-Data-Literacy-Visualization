# To Do: Create a visualization that will support your conclusions for part two
# NOTE: This file is a single-period snapshot (one row per agency per mode), not a
# monthly time series, so a literal "trend over years" isn't possible with this data.
# Instead, this chart breaks down ridership by mode of service per neighborhood,
# which reveals a different kind of pattern: how each area relies on different transit types.
mode_totals = df_filtered.groupby(["Agency", "Mode"])["Unlinked Passenger Trips FY"].sum().reset_index()

plt.figure(figsize=(12, 6))
sns.barplot(data=mode_totals, x="Mode", y="Unlinked Passenger Trips FY", hue="Agency")
plt.title("Ridership by Mode of Service Across Neighborhoods")
plt.xlabel("Mode of Service")
plt.ylabel("Unlinked Passenger Trips (FY)")
plt.xticks(rotation=45, ha="right")
plt.legend(title="Agency")
plt.tight_layout()

plt.savefig(os.path.join(output_folder, "ridership_by_mode.png"))
plt.show()
display(Image(filename=os.path.join(output_folder, "ridership_by_mode.png")))
